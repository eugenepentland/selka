        return False

        case Some(one):
            match pool.get(store, second):
                case Some(two):

                case None:
                    return False

            return False

public func test_recursive_record_equality_is_structural() -> bool:
    mut first = Node(value=1, next=None)
    mut second = Node(value=1, next=None)
    if first != second:
        return False
    if first != first:
        return False
    second.next = Some(first.copy())
    mut expected: Node? = Some(first)
    if second.next != expected:
        return False
    return True
