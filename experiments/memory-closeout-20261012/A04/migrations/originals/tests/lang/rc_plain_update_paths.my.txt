# A `mut` record callee that writes through a field path keeps the caller's
# receiver loan (RC-PLAIN-UPDATE-CALL-2). The receiver `frame.rows` is
# captured before the index call runs, so each read sees the captured list
# while the call's write reaches the caller. Answers match the collector;
# RC counting builds balance.
record Frame:
    rows: int[][]
    ip: int


func push(mut values: int[]):
    values.append(7)


func replace_first(mut rows: int[][]):
    rows[0] = [42]


func nested(mut frame: Frame) -> int:
    frame.rows[0].append(5)
    frame.ip = frame.ip + 1
    return 0


func field_argument(mut frame: Frame) -> int:
    replace_first(mut frame.rows)
    return 0


func element_argument(mut frame: Frame) -> int:
    push(mut frame.rows[0])
    return 0


func read_nested(mut frame: Frame) -> int:
    return len(frame.rows[nested(mut frame)])


func read_field_argument(mut frame: Frame) -> int:
    return frame.rows[field_argument(mut frame)][1]


func read_element_argument(mut frame: Frame) -> int:
    return len(frame.rows[element_argument(mut frame)])


public func test_a_nested_write_callee_keeps_the_captured_receiver() -> bool:
    mut frame = Frame(rows=[[1, 2], [3]], ip=0)
    mut seen = read_nested(mut frame)
    return seen == 2 and frame.rows == [[1, 2, 5], [3]] and frame.ip == 1


public func test_a_mut_field_argument_callee_keeps_the_captured_receiver() -> bool:
    mut frame = Frame(rows=[[1, 2], [3]], ip=0)
    mut seen = read_field_argument(mut frame)
    return seen == 2 and frame.rows == [[42], [3]]


public func test_a_mut_element_argument_callee_keeps_the_captured_receiver() -> bool:
    mut frame = Frame(rows=[[1, 2], [3]], ip=0)
    mut seen = read_element_argument(mut frame)
    return seen == 2 and frame.rows == [[1, 2, 7], [3]]


func main(out: Stream) -> int:
    out.print(test_a_nested_write_callee_keeps_the_captured_receiver())
    out.print(test_a_mut_field_argument_callee_keeps_the_captured_receiver())
    out.print(test_a_mut_element_argument_callee_keeps_the_captured_receiver())
    return 0
