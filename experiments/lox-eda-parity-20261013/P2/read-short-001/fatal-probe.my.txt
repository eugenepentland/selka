import chunk
import heap
import object
import table
import value
import vm_queries

func legacy(rt: heap.Runtime, mut frame: heap.CallFrame) -> int:
    frame.ip = frame.ip + 2
    return heap.function_instruction(rt, frame.function, frame.ip - 2) * 256 + heap.function_instruction(rt, frame.function, frame.ip - 1)

func fused(rt: heap.Runtime, mut frame: heap.CallFrame) -> int:
    frame.ip = frame.ip + 2
    return vm_queries.short_operand(rt, frame.function, frame.ip - 2, frame.ip - 1)

func main(args: str[], out: Stream) -> int:
    const mode = args[0].copy()
    const scenario = args[1].copy()
    mut rt = heap.new_runtime(False)
    mut function = object.new_function()
    mut code = chunk.new_chunk()
    mut ip = 0
    if scenario == "ip-overflow":
        ip = 9223372036854775807
    elif scenario == "high-bounds":
        pass
    elif scenario == "low-bounds":
        chunk.write_byte(mut code, 1, 1)
    elif scenario == "multiply-before-low-bounds":
        chunk.write_byte(mut code, 9223372036854775807, 1)
    elif scenario == "multiply-negative-overflow":
        chunk.write_byte(mut code, -36028797018963969, 1)
        chunk.write_byte(mut code, 0, 1)
    elif scenario == "add-overflow":
        chunk.write_byte(mut code, 36028797018963967, 1)
        chunk.write_byte(mut code, 256, 1)
    else:
        chunk.write_byte(mut code, 18, 1)
        chunk.write_byte(mut code, 52, 1)
    function.code = code
    mut ref = heap.allocate(mut rt, object.Obj.function(value=function))
    if scenario == "handle-bounds":
        ref = 1
    elif scenario == "kind-failure":
        ref = heap.allocate(mut rt, object.Obj.klass(value=object.ClassObj(name=value.NO_OBJ, methods=table.new_table())))
    mut frame = heap.CallFrame(closure=value.NO_OBJ, function=ref, ip=ip, base=0)
    mut result = 0
    if mode == "legacy":
        result = legacy(rt, mut frame)
    else:
        result = fused(rt, mut frame)
    _ = out.write(str(result).encode() + b"\n")
    return 0
