
import kicad_sch_textbox as m
import sexpr_parser as parser
import support_expect as ex
import byte_builder

func replace_bytes(src: bytes, old: bytes, new: bytes) -> bytes:
    mut w = byte_builder.create()
    mut pos = 0
    while pos < len(src):
        if src[pos:pos + len(old)] == old:
            byte_builder.append(mut w, new)
            pos += len(old)
        else:
            _ = byte_builder.append_byte(mut w, src[pos])
            pos += 1
    return byte_builder.finish(w)


func scan_text(src: bytes) -> m.Report?:
    match parser.parse(src):
        case Err(_):
            return None
        case Ok(nodes):
            if len(nodes) == 0:
                return None
            return Some(m.scan(nodes[0]))


# test "kicad-sch: text boxes follow justification and orientation"
public func test_kicad_sch_text_boxes_follow_justification_and_orientation(out: Stream) -> bool:
    mut right = m.box_of(m.Placement(x=0, y=0, dir=0, w=400, h=100, just=m.Just.start, cross=m.Cross.below))
    if not ex.eq_int(out, "right x0", right.x0, 0):
        return False
    if not ex.eq_int(out, "right x1", right.x1, 400):
        return False
    if not ex.eq_int(out, "right y0", right.y0, 0):
        return False
    if not ex.eq_int(out, "right y1", right.y1, 100):
        return False
    mut left = m.box_of(m.Placement(x=0, y=0, dir=180, w=400, h=100, just=m.Just.start, cross=m.Cross.below))
    if not ex.eq_int(out, "left x0", left.x0, -400):
        return False
    if not ex.eq_int(out, "left x1", left.x1, 0):
        return False
    mut up = m.box_of(m.Placement(x=0, y=0, dir=90, w=400, h=100, just=m.Just.start, cross=m.Cross.center))
    if not ex.eq_int(out, "up y0", up.y0, -400):
        return False
    if not ex.eq_int(out, "up y1", up.y1, 0):
        return False
    if not ex.eq_int(out, "up x0", up.x0, -50):
        return False
    if not ex.eq_int(out, "up x1", up.x1, 50):
        return False
    mut down = m.box_of(m.Placement(x=0, y=0, dir=270, w=400, h=100, just=m.Just.start, cross=m.Cross.center))
    if not ex.eq_int(out, "down y0", down.y0, 0):
        return False
    return ex.eq_int(out, "down y1", down.y1, 400)


# test "kicad-sch: the scan finds an overlapping label pair and clears a tidy sheet"
public func test_kicad_sch_the_scan_finds_an_overlapping_label_pair_and_clears_a_tidy_sheet(out: Stream) -> bool:
    mut clash = b"".join([b"(kicad_sch (lib_symbols)\n", b"(global_label \"VDD3V3\" (at 50 50 0) (effects (font (size 1.27 1.27)) (justify left)))\n", b"(global_label \"VDD3V3\" (at 50 50.5 0) (effects (font (size 1.27 1.27)) (justify left)))\n)\n"])
    mut hit = scan_text(clash)
    if hit is None:
        return False
    if not ex.eq_int(out, "texts", hit.texts, 2):
        return False
    if not ex.eq_int(out, "overlap", hit.overlaps, 1):
        return False
    mut first = hit.first.copy()
    if first is None:
        return False
    if not ex.eq_bytes(out, "first party", first.a, b"VDD3V3"):
        return False
    if not ex.ok(out, "label kind", first.kind_a == m.Kind.label):
        return False
    mut clear = replace_bytes(clash, b"(at 50 50.5 0)", b"(at 50 55 0)")
    mut ok = scan_text(clear)
    if ok is None:
        return False
    if not ex.eq_int(out, "clear texts", ok.texts, 2):
        return False
    if not ex.eq_int(out, "clear overlap", ok.overlaps, 0):
        return False
    return ex.ok(out, "no first collision", ok.first.is_none())


# test "kicad-sch: a zero-size pin name is not measured while its number still is"
public func test_kicad_sch_a_zero_size_pin_name_is_not_measured_while_its_number_still_is(out: Stream) -> bool:
    mut doc = b"".join([b"(kicad_sch (lib_symbols (symbol \"netlisp:BOX\" (pin_names (offset 0.254))\n", b"(symbol \"BOX_1_1\" (pin passive line (at -15.24 0 0) (length 2.54)\n", b"(name \"VSSAPM_U\" (effects (font (size 1.27 1.27))))\n", b"(number \"H14\" (effects (font (size 1.27 1.27))))))))\n", b"(symbol (lib_id \"netlisp:BOX\") (at 63.5 38.1 0) (unit 1)))\n"])
    mut shown = scan_text(doc)
    if shown is None:
        return False
    if not ex.eq_int(out, "name and number", shown.texts, 2):
        return False
    mut drawn = b"(name \"VSSAPM_U\" (effects (font (size 1.27 1.27))))"
    mut muted = b"(name \"VSSAPM_U\" (effects (font (size 0 0))))"
    mut quiet = replace_bytes(doc, drawn, muted)
    mut after = scan_text(quiet)
    if after is None:
        return False
    if not ex.eq_int(out, "number remains", after.texts, 1):
        return False
    return ex.eq_int(out, "no collision", after.overlaps, 0)


# P5: exact strict 10-unit boundaries, stable ties, and retained report strings.
public func test_p5_overlap_boundaries_ties_and_owned_report(out: Stream) -> bool:
    mut empty = scan_text(b"(kicad_sch)")
    if empty is None or empty.texts != 0 or empty.overlaps != 0 or empty.first.is_some():
        return False
    mut one = scan_text(b"(kicad_sch (global_label \"A\" (at -2.5 -2.5) (effects (font (size 1)))))")
    if one is None or one.texts != 1 or one.overlaps != 0:
        return False
    mut exact = scan_text(b"(kicad_sch (global_label \"A\" (at 0 0) (effects (font (size 1)))) (global_label \"B\" (at 0.9 0) (effects (font (size 1)))))")
    if exact is None or not ex.eq_int(out, "10-unit boundary", exact.overlaps, 0):
        return False
    mut inside = scan_text(b"(kicad_sch (global_label \"A\" (at 0 0) (effects (font (size 1)))) (global_label \"B\" (at 0.89 0) (effects (font (size 1)))))")
    if inside is None or not ex.eq_int(out, "11-unit overlap", inside.overlaps, 1):
        return False
    mut ties = scan_text(b"(kicad_sch (global_label \"A\" (at -2.5 -2.5) (effects (font (size 1)))) (global_label \"B\" (at -2.5 -2.5) (effects (font (size 1)))) (global_label \"C\" (at -2.5 -2.5) (effects (font (size 1)))))")
    if ties is None or not ex.eq_int(out, "dense ties", ties.overlaps, 3):
        return False
    match ties.first:
        case Some(pair):
            if not ex.eq_bytes(out, "stable first", pair.a, b"A") or not ex.eq_bytes(out, "stable second", pair.b, b"B"):
                return False
            return ex.eq_int(out, "negative x", pair.x, -138) and ex.eq_int(out, "negative y", pair.y, -300)
        case None:
            return False


public func test_p5_overlap_sparse_y_boundary_and_long_text(out: Stream) -> bool:
    mut clear = scan_text(b"(kicad_sch (global_label \"A\" (at 0 0) (effects (font (size 1)))) (global_label \"B\" (at 100 0) (effects (font (size 1)))) (global_label \"C\" (at 200 0) (effects (font (size 1)))))")
    if clear is None or clear.texts != 3 or clear.overlaps != 0:
        return False
    mut ybound = scan_text(b"(kicad_sch (global_label \"A\" (at 0 0) (effects (font (size 1)))) (global_label \"B\" (at 0 0.9) (effects (font (size 1)))))")
    if ybound is None or ybound.overlaps != 0:
        return False
    mut text = byte_builder.create()
    for i in range(4096):
        _ = byte_builder.append_byte(mut text, 65)
    mut long = byte_builder.finish(text)
    mut doc = b"".join([b"(kicad_sch (global_label \"", long.copy(), b"\" (at 0 0) (effects (font (size 1)))) (global_label \"B\" (at 0 0) (effects (font (size 1)))))"])
    mut hit = scan_text(doc)
    if hit is None or hit.texts != 2 or hit.overlaps != 1:
        return False
    match hit.first:
        case Some(pair):
            return ex.eq_bytes(out, "owned long report text", pair.a, long) and ex.eq_bytes(out, "owned short report text", pair.b, b"B")
        case None:
            return False


public func test_p5_overlap_sort_order_and_first_qualifying_at(out: Stream) -> bool:
    mut ordered = scan_text(b"(kicad_sch (global_label \"A\" (at 0.2 0) (effects (font (size 1)))) (global_label \"B\" (at 0 0.2) (effects (font (size 1)))) (global_label \"C\" (at 0 0) (effects (font (size 1)))))")
    if ordered is None or ordered.overlaps != 3:
        return False
    match ordered.first:
        case Some(pair):
            if pair.a != b"C" or pair.b != b"B":
                return False
        case None:
            return False
    mut short = scan_text(b"(kicad_sch (global_label \"A\" (at 0) (at 0 0) (effects (font (size 1)))))")
    if short is None or not ex.eq_int(out, "short at is skipped", short.texts, 1):
        return False
    mut invalid = scan_text(b"(kicad_sch (global_label \"A\" (at bad 0) (at 0 0) (effects (font (size 1)))))")
    if invalid is None:
        return False
    return ex.eq_int(out, "first full at is selected", invalid.texts, 0)


func p5_followup_named_doc(value: bytes) -> bytes:
    mut doc = b"".join([b"(kicad_sch (lib_symbols (symbol \"netlisp:BOX\" (pin_names (offset 0.254))\n", b"(symbol \"BOX_1_1\" (pin passive line (at -15.24 0 0) (length 2.54)\n", b"(name ", value.copy(), b" (effects (font (size 1.27 1.27))))\n", b"(number \"H14\" (effects (font (size 0 0))))))))\n", b"(symbol (lib_id \"netlisp:BOX\") (at 63.5 38.1 0) (unit 1)))\n"])
    mut placement = b"(symbol (lib_id \"netlisp:BOX\") (at 63.5 38.1 0) (unit 1))\n"
    return replace_bytes(doc, b"(symbol (lib_id \"netlisp:BOX\") (at 63.5 38.1 0) (unit 1)))\n", placement + placement + b")\n")


# test "P5 follow-up: named pin atom/string ownership survives AST destruction"
public func test_p5_followup_named_pin_owned_values(out: Stream) -> bool:
    for input in [b"ALPHA", b"\"ALPHA\""]:
        mut report = scan_text(p5_followup_named_doc(input))
        if report is None or report.texts != 2 or report.overlaps != 1 or report.first is None:
            return False
        mut want = b"ALPHA"
        if not p5_followup_pair_is(report, want):
            return False
    mut long = b"".join([for i in range(512) yield b"ABCDEFGH"])
    mut report = scan_text(p5_followup_named_doc(b"\"" + long + b"\""))
    return report is not None and report.first is not None and p5_followup_pair_is(report, long)


# test "P5 follow-up: unsupported named pin nodes preserve empty fallback"
public func test_p5_followup_named_pin_empty_fallback() -> bool:
    for input in [b"7", b"1.5", b"()"]:
        mut report = scan_text(p5_followup_named_doc(input))
        if report is None or report.texts != 0 or report.overlaps != 0 or report.first is not None:
            return False
    return True


# test "P5 follow-up: library selection retains the first duplicate ID"
public func test_p5_followup_first_library_match() -> bool:
    mut original = p5_followup_named_doc(b"FIRST")
    mut duplicate = b"".join([b"(lib_symbols (symbol \"netlisp:BOX\" (symbol \"BOX_1_1\" ", b"(pin passive line (at -15.24 0 0) (length 2.54) (name \"WRONG\" ", b"(effects (font (size 1.27 1.27)))) (number \"H14\" (effects (font (size 0 0))))))))\n"])
    mut doc = replace_bytes(original, b"(symbol (lib_id", duplicate + b"(symbol (lib_id")
    mut report = scan_text(doc)
    return report is not None and report.texts == 2 and report.overlaps == 1 and report.first is not None and p5_followup_pair_is(report, b"FIRST")


func p5_followup_pair_is(report: m.Report, want: bytes) -> bool:
    match report.first:
        case Some(pair):
            return pair.a == want and pair.b == want
        case None:
            return False
