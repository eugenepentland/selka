module main

# Probe 1: KiCad-style S-expression parser written the way a Python programmer would.

record Pos:
    offset: int
    line: int
    col: int

record ParseError:
    message: str
    pos: Pos

union Node:
    atom: str,
    text: str,
    items: list[Node]

class Cursor:
    src: str
    pos: int
    line: int
    col: int

def here(c: Cursor) -> Pos:
    return Pos(offset=c.pos, line=c.line, col=c.col)

def peek(c: Cursor) -> str:
    if c.pos >= len(c.src):
        return ""
    return c.src[c.pos]

def advance(c: Cursor) -> None:
    ch = c.src[c.pos]
    c.pos = c.pos + 1
    if ch == "\n":
        c.line = c.line + 1
        c.col = 1
    else:
        c.col = c.col + 1

def skip_ws(c: Cursor) -> None:
    while c.pos < len(c.src):
        ch = peek(c)
        if ch.isspace():
            advance(c)
        elif ch == "#":
            while c.pos < len(c.src) and peek(c) != "\n":
                advance(c)
        else:
            break

def fail(c: Cursor, message: str) -> ParseError:
    return ParseError(message=message, pos=here(c))

def parse_string(c: Cursor) -> Result[Node, ParseError]:
    advance(c)  # opening quote
    out = ""
    while True:
        if c.pos >= len(c.src):
            return Err(fail(c, "unterminated string"))
        ch = peek(c)
        if ch == '"':
            advance(c)
            return Ok(Node.text(out))
        if ch == "\\":
            advance(c)
            esc = peek(c)
            if esc == "n":
                out = out + "\n"
            elif esc == "t":
                out = out + "\t"
            else:
                out = out + esc
            advance(c)
        else:
            out = out + ch
            advance(c)

def parse_atom(c: Cursor) -> Result[Node, ParseError]:
    start = c.pos
    while c.pos < len(c.src):
        ch = peek(c)
        if ch.isspace() or ch == "(" or ch == ")":
            break
        advance(c)
    if start == c.pos:
        return Err(fail(c, "expected atom"))
    return Ok(Node.atom(c.src[start:c.pos]))

def parse_node(c: Cursor) -> Result[Node, ParseError]:
    skip_ws(c)
    ch = peek(c)
    if ch == "":
        return Err(fail(c, "unexpected end of input"))
    if ch == "(":
        return parse_list(c)
    if ch == '"':
        return parse_string(c)
    if ch == ")":
        return Err(fail(c, "unexpected ')'"))
    return parse_atom(c)

def parse_list(c: Cursor) -> Result[Node, ParseError]:
    advance(c)  # (
    items: list[Node] = []
    while True:
        skip_ws(c)
        if peek(c) == ")":
            advance(c)
            return Ok(Node.items(items))
        if peek(c) == "":
            return Err(fail(c, "unclosed list"))
        node = parse_node(c)?
        items.append(node)

def parse(src: str) -> Result[Node, ParseError]:
    c = Cursor(src=src, pos=0, line=1, col=1)
    node = parse_node(c)?
    skip_ws(c)
    if c.pos != len(c.src):
        return Err(fail(c, "trailing input"))
    return Ok(node)

# --- queries -----------------------------------------------------------

def head(node: Node) -> str:
    match node:
        case Node.items(xs):
            if len(xs) == 0:
                return ""
            return head(xs[0])
        case Node.atom(a):
            return a
        case Node.text(t):
            return t

def children(node: Node, name: str) -> list[Node]:
    out: list[Node] = []
    match node:
        case Node.items(xs):
            for x in xs:
                match x:
                    case Node.items(inner):
                        if head(x) == name:
                            out.append(x)
                    case _:
                        pass
        case _:
            pass
    return out

def nth_atom(node: Node, index: int) -> str:
    match node:
        case Node.items(xs):
            if index < len(xs):
                return head(xs[index])
            return ""
        case _:
            return ""

def render(node: Node) -> str:
    match node:
        case Node.atom(a):
            return a
        case Node.text(t):
            return '"' + t + '"'
        case Node.items(xs):
            parts = [render(x) for x in xs]
            return "(" + " ".join(parts) + ")"

def count_nodes(node: Node) -> int:
    match node:
        case Node.items(xs):
            total = 1
            for x in xs:
                total = total + count_nodes(x)
            return total
        case _:
            return 1

def main(args: list[str], fs: Filesystem, out: Stream) -> int:
    src = ""
    match read_text(fs, args[0]):
        case Ok(text):
            src = text
        case Err(problem):
            out.print("read failed", problem)
            return 2
    match parse(src):
        case Ok(node):
            out.print(count_nodes(node), len(children(node, "footprint")))
            return 0
        case Err(e):
            out.print("error", e.message, e.pos.line, e.pos.col)
            return 1
